Query::is_post_edit()			|| ! Post_Type::is_supported( $post_type )			|| ! \apply_filters( 'better_seo_seobox_output', true )		) {			return;		}		$box_id = 'better-seo-inpost-box';		\add_meta_box(			$box_id,			\esc_html__( 'SEO Settings', 'better-seo' ),			[ self::class, 'meta_box' ],			null,			/**			 * Filters the Better SEO meta box context.			 *			 * @since 1.0.0			 * @param string $context Accepts 'normal', 'side', and 'advanced'.			 */			(string) \apply_filters( 'better_seo_metabox_context', 'normal' ),			/**			 * Filters the Better SEO meta box priority.			 *			 * @since 1.0.0			 * @param string $priority Accepts 'high', 'default', 'low'.			 */			(string) \apply_filters( 'better_seo_metabox_priority', 'high' ),		);		$screen_id = \get_current_screen()->id;		\add_filter( "postbox_classes_{$screen_id}_{$box_id}", [ self::class, 'add_postbox_class' ] );		if ( ! is_headless( 'settings' ) && Query::is_static_front_page( Query::get_the_real_id() ) ) {			$output_homepage_warning = [ self::class, 'output_homepage_warning' ];			\add_action( 'better_seo_pre_page_inpost_general_tab', $output_homepage_warning );			\add_action( 'better_seo_pre_page_inpost_visibility_tab', $output_homepage_warning );			\add_action( 'better_seo_pre_page_inpost_social_tab', $output_homepage_warning );		}	}	/**	 * Outputs the flex nav tab wrapper for the post settings meta box.	 *	 * @since 1.0.0	 *	 * @param string             $id   The wrapper ID.	 * @param array<string, mixed> $tabs The tab definitions.	 * @return void	 */	public static function flex_nav_tab_wrapper( string $id, array $tabs = [] ): void {		Template::output_view( 'post/wrap-nav', $id, $tabs );		Template::output_view( 'post/wrap-content', $id, $tabs );	}	/**	 * Outputs the Better SEO meta box content.	 *	 * @since 1.0.0	 *	 * @return void	 */	public static function meta_box(): void {		\wp_nonce_field(			Data\Admin\Post::SAVE_NONCES['post-edit']['action'],			Data\Admin\Post::SAVE_NONCES['post-edit']['name'],		);		/**		 * Fires before the Better SEO post inpost box is rendered.		 *		 * @since 1.0.0		 */		\do_action( 'better_seo_pre_page_inpost_box' );		if ( Query::is_block_editor() ) {			Template::output_view( 'post/gutenberg-data' );		}		Template::output_view( 'post/settings', 'main' );		/**		 * Fires after the Better SEO post inpost box is rendered.		 *		 * @since 1.0.0		 */		\do_action( 'better_seo_pro_page_inpost_box' );	}	/**	 * Adds the Better SEO block editor class to the meta box postbox classes.	 *	 * @since 1.0.0	 *	 * @param array<int, string> $classes Existing postbox CSS classes.	 * @return array<int, string> Modified postbox CSS classes.	 */	public static function add_postbox_class( array $classes = [] ): array {		if ( Query::is_block_editor() ) {			$classes[] = 'better-seo-is-block-editor';		}		return $classes;	}	/**	 * Outputs the homepage warning template.	 *	 * @since 1.0.0	 *	 * @return void	 */	public static function output_homepage_warning(): void {		Template::output_view( 'post/homepage-warning' );	}	/**	 * Outputs the General tab content for the post settings meta box.	 *	 * @since 1.0.0	 *	 * @return void	 */	public static function general_tab(): void {		/**		 * Fires before the Better SEO general tab is rendered.		 *		 * @since 1.0.0		 */		\do_action( 'better_seo_pre_page_inpost_general_tab' );		Template::output_view( 'post/settings', 'general' );		/**		 * Fires after the Better SEO general tab is rendered.		 *		 * @since 1.0.0		 */		\do_action( 'better_seo_pro_page_inpost_general_tab' );	}	/**	 * Outputs the Visibility tab content for the post settings meta box.	 *	 * @since 1.0.0	 *	 * @return void	 */	public static function visibility_tab(): void {		/**		 * Fires before the Better SEO visibility tab is rendered.		 *		 * @since 1.0.0		 */		\do_action( 'better_seo_pre_page_inpost_visibility_tab' );		Template::output_view( 'post/settings', 'visibility' );		/**		 * Fires after the Better SEO visibility tab is rendered.		 *		 * @since 1.0.0		 */		\do_action( 'better_seo_pro_page_inpost_visibility_tab' );	}	/**	 * Outputs the Social tab content for the post settings meta box.	 *	 * @since 1.0.0	 *	 * @return void	 */	public static function social_tab(): void {		/**		 * Fires before the Better SEO social tab is rendered.		 *		 * @since 1.0.0		 */		\do_action( 'better_seo_pre_page_inpost_social_tab' );		Template::output_view( 'post/settings', 'social' );		/**		 * Fires after the Better SEO social tab is rendered.		 *		 * @since 1.0.0		 */		\do_action( 'better_seo_pro_page_inpost_social_tab' );	}}
