	/**
	 * Outputs the SEO Settings page wrap.
	 *
	 * @hook toplevel_page_theseoframework-settings 10
	 * @since 4.0.0
	 * @since 5.0.0 Moved from `\BETTER_SEO\Load`.
	 * @since 5.0.1 Removed the registration of metaboxes.
	 */
	public static function prepare_settings_wrap() {

		\add_action(
			Admin\Menu::get_page_hook_name() . '_settings_page_boxes',
			[ self::class, 'output_columns' ],
		);
		\add_action(
			'BETTER_SEO_setting_notices',
			[ self::class, 'output_notices' ],
		);

		self::output_wrap();
	}

		public static function register_seo_settings_meta_boxes() {

				$general           = (bool) \apply_filters( 'BETTER_SEO_general_metabox', true );
		$title             = (bool) \apply_filters( 'BETTER_SEO_title_metabox', true );
		$description       = (bool) \apply_filters( 'BETTER_SEO_description_metabox', true );
		$robots            = (bool) \apply_filters( 'BETTER_SEO_robots_metabox', true );
		$home              = (bool) \apply_filters( 'BETTER_SEO_home_metabox', true );
		$post_type_archive = (bool) \apply_filters( 'BETTER_SEO_post_type_archive_metabox', true );
		$social            = (bool) \apply_filters( 'BETTER_SEO_social_metabox', true );
		$schema            = (bool) \apply_filters( 'BETTER_SEO_schema_metabox', true );
		$webmaster         = (bool) \apply_filters( 'BETTER_SEO_webmaster_metabox', true );
		$sitemap           = (bool) \apply_filters( 'BETTER_SEO_sitemap_metabox', true );
		$feed              = (bool) \apply_filters( 'BETTER_SEO_feed_metabox', true );

		$settings_page_hook = Admin\Menu::get_page_hook_name();

		// General Meta Box
		if ( $general )
			\add_meta_box(
				'autodescription-general-settings',
				\esc_html__( 'General Settings', 'autodescription' ),
				[ self::class, '_general_metabox' ],
				$settings_page_hook,
				'main',
			);

		// Title Meta Box
		if ( $title )
			\add_meta_box(
				'autodescription-title-settings',
				\esc_html__( 'Title Settings', 'autodescription' ),
				[ self::class, '_title_metabox' ],
				$settings_page_hook,
				'main',
			);

		// Description Meta Box
		if ( $description )
			\add_meta_box(
				'autodescription-description-settings',
				\esc_html__( 'Description Meta Settings', 'autodescription' ),
				[ self::class, '_description_metabox' ],
				$settings_page_hook,
				'main',
			);

		// Social Meta Box
		if ( $social )
			\add_meta_box(
				'autodescription-social-settings',
				\esc_html__( 'Social Meta Settings', 'autodescription' ),
				[ self::class, '_social_metabox' ],
				$settings_page_hook,
				'main',
			);

		// Homepage Meta Box
		if ( $home )
			\add_meta_box(
				'autodescription-homepage-settings',
				\esc_html__( 'Homepage Settings', 'autodescription' ),
				[ self::class, '_homepage_metabox' ],
				$settings_page_hook,
				'main',
			);

		if ( $post_type_archive && Post_Type::get_public_pta() )
			\add_meta_box(
				'autodescription-post-type-archive-settings',
				\esc_html__( 'Post Type Archive Settings', 'autodescription' ),
				[ self::class, '_post_type_archive_metabox' ],
				$settings_page_hook,
				'main',
			);

		// Schema Meta Box
		if ( $schema )
			\add_meta_box(
				'autodescription-schema-settings',
				\esc_html__( 'Schema.org Settings', 'autodescription' ),
				[ self::class, '_schema_metabox' ],
				$settings_page_hook,
				'main',
			);

		// Robots Meta Box
		if ( $robots )
			\add_meta_box(
				'autodescription-robots-settings',
				\esc_html__( 'Robots Settings', 'autodescription' ),
				[ self::class, '_robots_metabox' ],
				$settings_page_hook,
				'main',
			);

		// Webmaster Meta Box
		if ( $webmaster )
			\add_meta_box(
				'autodescription-webmaster-settings',
				\esc_html__( 'Webmaster Meta Settings', 'autodescription' ),
				[ self::class, '_webmaster_metabox' ],
				$settings_page_hook,
				'main',
			);

		// Sitemaps Meta Box
		if ( $sitemap )
			\add_meta_box(
				'autodescription-sitemap-settings',
				\esc_html__( 'Sitemap Settings', 'autodescription' ),
				[ self::class, '_sitemaps_metabox' ],
				$settings_page_hook,
				'main',
			);

		// Feed Meta Box
		if ( $feed )
			\add_meta_box(
				'autodescription-feed-settings',
				\esc_html__( 'Feed Settings', 'autodescription' ),
				[ self::class, '_feed_metabox' ],
				$settings_page_hook,
				'main',
			);
	}

	/**
	 * Outputs SEO Settings page wrap.
	 *
	 */
	public static function output_wrap() {
		/**
		 * @since 3.0.0
		 */
		\do_action( 'BETTER_SEO_pre_seo_settings' );
		Template::output_view( 'settings/wrap' );
		/**
		 * @since 3.0.0
		 */
		\do_action( 'BETTER_SEO_pro_seo_settings' );
	}
